cmake_minimum_required(VERSION 3.16)
project(uav_simulation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(JPEG REQUIRED)
add_executable(gazebo_receiver gazebo_receiver.cpp)
target_compile_features(gazebo_receiver PRIVATE cxx_std_17)
target_include_directories(gazebo_receiver PRIVATE ../bridge/include)
target_link_libraries(gazebo_receiver JPEG::JPEG)
ament_target_dependencies(gazebo_receiver rclcpp nav_msgs sensor_msgs rosgraph_msgs)
install(TARGETS gazebo_receiver DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE
  PATTERN "*.pyc" EXCLUDE)
ament_package()
